/*:
 * @plugindesc Changes HP/MP label colors in battle UI (HP: #b0ff90, MP: #b83dba)
 * @author ChatGPT
 *
 * @help
 * Lightweight plugin that recolors HP and MP label text in battle/status windows.
 */

(function() {

    const HP_COLOR = "#b0ff90";
    const MP_COLOR = "#b83dba";

    const _drawActorHpLabel = Window_StatusBase.prototype.drawActorHpLabel;
    Window_StatusBase.prototype.drawActorHpLabel = function(actor, x, y, width) {
        const c = this.contents.textColor;
        this.changeTextColor(HP_COLOR);
        _drawActorHpLabel.call(this, actor, x, y, width);
        this.changeTextColor(c);
    };

    const _drawActorMpLabel = Window_StatusBase.prototype.drawActorMpLabel;
    Window_StatusBase.prototype.drawActorMpLabel = function(actor, x, y, width) {
        const c = this.contents.textColor;
        this.changeTextColor(MP_COLOR);
        _drawActorMpLabel.call(this, actor, x, y, width);
        this.changeTextColor(c);
    };

})();
